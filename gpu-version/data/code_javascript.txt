// 5. 반복문 / loops
function sumToN(n) {
  let total = 0;
  for (let i = 1; i <= n; i++) {
    total += i;
  }
  return total;
}

function countdown(n) {
  while (n > 0) {
    console.log(n);
    n--;
  }
  console.log("발사!");
}

function findMax(items) {
  let biggest = items[0];
  for (const item of items) {
    if (item > biggest) biggest = item;
  }
  return biggest;
}

// 7. 재귀 / recursion
function factorial(n) {
  if (n <= 1) return 1;
  return n * factorial(n - 1);
}

function fibonacci(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

// 8. 문자열 처리 / string handling
function reverseString(s) {
  return s.split("").reverse().join("");
}

function isPalindrome(s) {
  const cleaned = s.replace(/\s/g, "").toLowerCase();
  return cleaned === cleaned.split("").reverse().join("");
}

// 9. 클래스 / classes
class Player {
  constructor(name, hp = 100) {
    this.name = name;
    this.hp = hp;
    this.level = 1;
  }

  takeDamage(amount) {
    this.hp -= amount;
    if (this.hp < 0) this.hp = 0;
  }

  isAlive() {
    return this.hp > 0;
  }

  levelUp() {
    this.level += 1;
    this.hp += 10;
  }
}

class Inventory {
  constructor() {
    this.items = [];
  }
  addItem(item) {
    this.items.push(item);
  }
  removeItem(item) {
    this.items = this.items.filter((i) => i !== item);
  }
}

// 12. 간단한 캔버스 게임 루프 / simple canvas game loop
function gameLoop(ctx, player) {
  ctx.clearRect(0, 0, 800, 600);
  ctx.fillRect(player.x, player.y, 32, 32);
  requestAnimationFrame(() => gameLoop(ctx, player));
}

// 13. 정렬 알고리즘 / sorting
function bubbleSort(arr) {
  const n = arr.length;
  for (let i = 0; i < n; i++) {
    for (let j = 0; j < n - i - 1; j++) {
      if (arr[j] > arr[j + 1]) {
        [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
      }
    }
  }
  return arr;
}

// 14. 소수 판별 / prime check
function isPrime(n) {
  if (n < 2) return false;
  for (let i = 2; i * i <= n; i++) {
    if (n % i === 0) return false;
  }
  return true;
}
